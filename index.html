function mostrarSecoes(nomeLocal, zona) { const div = document.getElementById("resultadoLocal"); const registros = dados.filter(d => d.zonaNova === zona && d.local === nomeLocal); if (registros.length === 0) { div.innerHTML = '

Nenhuma seção encontrada.

'; return; } const endereco = registros[0].endereco; // Pega apenas seções ÚNICAS (sem repetir) const secoesUnicas = [...new Set(registros.map(r => r.secaoNova))] .sort((a, b) => Number(a) - Number(b)); let html = `

📍 ${nomeLocal}

Endereço: ${endereco}

Zona: ${zona}

Seções (${secoesUnicas.length}):

${secoesUnicas.map(s => `${s}`).join("")}
`; div.innerHTML = html; }
